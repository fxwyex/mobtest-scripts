   function setupKeybindInput(element, callback) {

       element.addEventListener("keydown", function (e) {
           if (keybindMasterToggle == false) return;
           if (CustomKeys[e.code] != undefined) {
               CustomKeys[e.code] = true;
           }
           e.preventDefault();
           if (e.key === "Escape") {
               element.value = "";
               if (callback) callback(null);
               return;
           }

           let keys = [];
           if (e.ctrlKey) keys.push("Ctrl");
           if (e.shiftKey) keys.push("Shift");
           if (e.altKey) keys.push("Alt");
           if (e.metaKey) keys.push("Meta"); // CMD on Mac
           if (CustomKeys["Space"] == true) keys.push("Space")
           if (keys.length > 0) e.preventDefault();
           let key = normalizeKey(e);

           if (!isModifierKey(e.key)) {
               keys.push(key);
           }

           let combo = keys.join("+");
           element.value = combo;

           if (callback) callback(combo);
       });
   }

    function normalizeKey(e) {
        if (e.code.startsWith("Numpad")) {
            return e.code.replace("Numpad", "Numpad ");
        }
        if (e.key === " ") return "Space";
        if (e.key.startsWith("Arrow")) {
            return e.key.replace("Arrow", "");
        }
        if (e.key.length === 1) {
            return e.key.toUpperCase();
        }
        return e.key;
    }

            key === "Alt" ||
            key === "Meta" ||
            key === " "
        );
    }
